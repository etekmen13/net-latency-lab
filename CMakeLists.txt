cmake_minimum_required(VERSION 3.20)
project(net_latency_lab LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(NLL_PI4 "Enable Raspberry Pi 4 Cortex-A72 tuning" OFF)
option(NLL_TSAN "Build with ThreadSanitizer" OFF)
option(BUILD_TESTING "Build native tests (requires system GTest)" OFF)

add_library(nll_options INTERFACE)
target_include_directories(nll_options INTERFACE "${CMAKE_SOURCE_DIR}/src")
target_compile_options(nll_options INTERFACE -Wall -Wextra -Wpedantic -fno-omit-frame-pointer)
if(NLL_PI4)
  target_compile_options(nll_options INTERFACE -mcpu=cortex-a72)
endif()
if(NLL_TSAN)
  target_compile_options(nll_options INTERFACE -fsanitize=thread -O1 -g)
  target_link_options(nll_options INTERFACE -fsanitize=thread)
endif()

foreach(target receiver_baseline receiver_batched receiver_threaded)
  add_executable(${target} "src/receiver/${target}.cpp")
  target_link_libraries(${target} PRIVATE nll_options pthread)
endforeach()
add_executable(sender src/sender/sender.cpp)
target_link_libraries(sender PRIVATE nll_options pthread)

if(BUILD_TESTING)
  include(CTest)
  find_package(GTest CONFIG REQUIRED)
  add_executable(native_tests tests/native_tests.cpp)
  target_link_libraries(native_tests PRIVATE nll_options GTest::gtest_main pthread)
  include(GoogleTest)
  gtest_discover_tests(native_tests)
  if(NLL_TSAN AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
    find_program(SETARCH_EXECUTABLE setarch)
  endif()
  if(SETARCH_EXECUTABLE)
    add_test(NAME spsc_stress COMMAND ${SETARCH_EXECUTABLE} x86_64 -R
      $<TARGET_FILE:native_tests> --gtest_filter=SPSCQueue.ConcurrentOrderedTransfers:SPSCQueue.ShutdownDrainsPublishedData)
  else()
    add_test(NAME spsc_stress COMMAND native_tests --gtest_filter=SPSCQueue.ConcurrentOrderedTransfers:SPSCQueue.ShutdownDrainsPublishedData)
  endif()
  set_tests_properties(spsc_stress PROPERTIES LABELS "spsc;tsan")
endif()
